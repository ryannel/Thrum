//! Capture the no-demand cooked valley before the M2 changes.
use std::path::PathBuf;
use thrum_render::{capabilities::HardwareRequest, present::SurfaceProfile};
use v1_game::{FrameConfig, FrameHost, GameSetup, RouteOverlay};

#[test]
fn reference() {
    let gpu = thrum_test_support::gpu("main valley reference", Some(wgpu::Backend::Metal)).expect("Metal");
    let request = HardwareRequest::parse("ray-query").expect("explicit hardware");
    let (device, queue, caps) = gpu.fresh_adapter().device_with_hardware(
        thrum_test_support::Profile::Portable, &thrum_test_support::Restriction::default(),
        1, &request
    ).expect("device");
    let caps = caps.with_recipe_override("apple-tile").expect("recipe");
    let root = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
    let store = root.join("../../reference-store");
    let config = thrum_cook::CookConfig {
        source: root.join("assets/showcase"), store: store.clone(), profiles: vec![thrum_asset::Profile::Portable],
        quality: thrum_cook::Quality::Fast, durability: thrum_cook::Durability::Process,
    };
    assert!(thrum_cook::Cooker::open(config).unwrap().cook_all().unwrap().failed.is_empty());
    let extent = [320, 180];
    let mut host = pollster::block_on(FrameHost::new_with_setup(&device, &queue, &caps, FrameConfig {
        render_extent: extent, surface: Some(SurfaceProfile { format: wgpu::TextureFormat::Bgra8Unorm, extent }),
    }, GameSetup { route_overlay: RouteOverlay::Proof, ..GameSetup::default() })).unwrap();
    let (camera, assets) = v1_game::prepare_cooked_valley(store.into()).unwrap();
    let stats = assets.stats();
    host.set_scene_contribution(Box::new(assets)).unwrap();
    host.set_scene_camera(camera).unwrap();
    host.set_time_of_day(0.5).unwrap();
    host.pause().unwrap();
    let limit = std::time::Instant::now() + std::time::Duration::from_secs(30);
    loop {
        let bytes = host.capture_pixels().unwrap();
        let stats = stats.lock().unwrap();
        assert!(stats.errors.is_empty());
        if stats.drawn.is_some() {
            std::fs::write(std::env::var_os("THRUM_RAY_DEMAND_BYTES").expect("output path"), bytes)
                .expect("write main's submitted bytes");
            break;
        }
        assert!(std::time::Instant::now() < limit);
    }
}
